a) sendKeys() -- Enters the specified text into an input field or text area.

b) clear() -- Clears the existing text from an input field.

c) click() -- Performs a mouse click action on a web element such as a button, link, checkbox, or radio button.

d) findElement() -- Locates and returns the first matching web element using the specified locator.

e) findElements() -- Locates and returns a list of all matching web elements using the specified locator.

f) getDomAttribute() -- Returns the value of the specified attribute directly from the element’s DOM.

g) getScreenshotAs() -- Captures a screenshot of the current element or page and returns it in the specified format.

h) getText() -- Returns the visible text present on the web element.

i) isDisplayed() -- Checks whether the web element is visible on the web page.

j) isEnabled() -- Checks whether the web element is enabled for user interaction.

k) isSelected() -- Checks whether a checkbox, radio button, or option is currently selected.
